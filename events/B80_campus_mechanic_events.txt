# Balanced semester.
country_event = {
	id = B80_campus.11
	title = B80_campus.11.t
	desc = B80_campus.11.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	option = {
		name = B80_campus.11.a
		add_political_power = 75
		add_stability = 0.03
		add_to_variable = { B80_school_reputation = 15 }
		add_timed_idea = { idea = B80_balanced_semester days = 120 }
		if = {
			limit = { has_country_flag = B80_opponent_system_initialized }
			subtract_from_variable = { B80_expansion_alarm = 3 }
			B80_clamp_opponent_values = yes
		}
		if = {
			limit = {
				check_variable = { B80_balanced_semester_streak > 2 }
				NOT = { has_idea = B80_wangjing_education_model }
			}
			add_ideas = B80_wangjing_education_model
			add_research_slot = 1
			add_to_variable = { B80_school_reputation = 20 }
		}
		B80_begin_new_semester = yes
	}
	option = {
		name = B80_campus.11.b
		add_to_variable = { B80_school_reputation = 8 }
		add_breakthrough_progress = { specialization = specialization_land value = 0.025 }
		add_breakthrough_progress = { specialization = specialization_air value = 0.025 }
		add_breakthrough_progress = { specialization = specialization_naval value = 0.025 }
		add_breakthrough_progress = { specialization = specialization_nuclear value = 0.025 }
		add_tech_bonus = { name = B80_balanced_term_bonus bonus = 0.50 uses = 2 category = electronics }
		if = {
			limit = { has_country_flag = B80_opponent_system_initialized }
			subtract_from_variable = { B80_expansion_alarm = 3 }
			B80_clamp_opponent_values = yes
		}
		B80_begin_new_semester = yes
	}
}
